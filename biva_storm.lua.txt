-- Hàm mã hóa URL an toàn (thay thế HttpService:UrlEncode)
local function url_encode(str)
    if type(str) ~= "string" then return "" end
    return (str:gsub("([^%w ])", function(c)
        return string.format("%%%02X", string.byte(c))
    end):gsub(" ", "%%20"))
end

-- FIX HWID: Lưu vào file để không mất key khi đổi tên
local HWID_FILE = "BivaStorm_HWID.txt"
local hwid
pcall(function()
    if isfile and readfile then
        if isfile(HWID_FILE) and readfile(HWID_FILE) ~= "" then
            hwid = readfile(HWID_FILE)
        else
            hwid = tostring(player.UserId)
            writefile(HWID_FILE, hwid)
        end
    else
        hwid = tostring(player.UserId)
    end
end)

local dragging, dragInput, dragStart, startPos
f.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = f.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)
f.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)
UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        f.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- LOGIC KÍCH HOẠT
btn.MouseButton1Click:Connect(function()
    local cleanKey = string.gsub(tb.Text, "%s+", "")
    if cleanKey == "" then
        status.TextColor3 = Color3.fromRGB(255, 65, 110)
        status.Text = "⚠️ Vui lòng nhập Key trước khi kích hoạt!"
        return
    end
    btn.Text = "⏳ ĐANG TẢI..."
    btn.BackgroundColor3 = Color3.fromRGB(100, 110, 125)
    status.TextColor3 = Color3.fromRGB(255, 255, 255)
    status.Text = "Đang kết nối tới máy chủ Biva Storm..."
    
    task.spawn(function()
        local url = "https://biva-hub.onrender.com/api/get_script?key="
            .. url_encode(cleanKey)
            .. "&hwid=" .. url_encode(hwid)
            .. "&game_id=" .. url_encode(selectedGameId)
            
        local success, code = pcall(function()
            return game:HttpGet(url)
        end)
        
        if success and code then
            if string.find(code, "Kick") then
                loadstring(code)()
            else
                local func, err = loadstring(code)
                if func then
                    pcall(function()
                        if writefile then
                            writefile(SAVE_KEY_FILE, cleanKey)
                        end
                    end)
                    sg:Destroy()
                    func()
                else
                    btn.Text = "⚡ KÍCH HOẠT SCRIPT"
                    btn.BackgroundColor3 = Color3.fromRGB(0, 255, 204)
                    status.TextColor3 = Color3.fromRGB(255, 65, 110)
                    status.Text = "Lỗi Code: " .. tostring(err)
                end
            end
        else
            btn.Text = "⚡ KÍCH HOẠT SCRIPT"
            btn.BackgroundColor3 = Color3.fromRGB(0, 255, 204)
            status.TextColor3 = Color3.fromRGB(255, 65, 110)
            status.Text = "❌ Lỗi kết nối Server!"
        end
    end)
end)
